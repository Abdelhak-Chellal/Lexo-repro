module.exports = function(input_path, new_extension) {
    if (input_path === null) return null;
    if (typeof input_path !== 'string') return input_path;

    if (input_path === '') {
        return new_extension === '' ? '' : '.' + new_extension;
    }

    if (new_extension === '') {
        const lastDotIndex = input_path.lastIndexOf('.');
        return lastDotIndex === -1 ? input_path : input_path.substring(0, lastDotIndex);
    } else {
        if (!new_extension.startsWith('.')) {
            new_extension = '.' + new_extension;
        }
    }

    const lastDotIndex = input_path.lastIndexOf('.');
    if (lastDotIndex === -1) {
        return input_path + new_extension;
    }

    const basePath = input_path.substring(0, lastDotIndex);
    return basePath + new_extension;
};
