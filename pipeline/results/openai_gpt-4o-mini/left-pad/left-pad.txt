module.exports = function(s, n, pad_char) {
    if (s === null) {
        s = 'null';
    } else if (typeof s === 'boolean') {
        s = String(s);
    } else {
        s = String(s);
    }

    const len_s = s.length;

    if (n <= len_s) {
        return s;
    }

    if (pad_char === null || pad_char === '') {
        pad_char = ' ';
    }

    const padding_length = n - len_s;
    const padding = pad_char.repeat(padding_length);

    return padding + s;
};
