```javascript
module.exports = function(value, length, padChar) {
    if (value === null) {
        value = 'null';
    } else {
        value = String(value);
    }

    if (padChar === null) {
        padChar = ' ';
    }

    if (length <= 0) {
        return '';
    }

    const currentLength = value.length;

    if (currentLength >= length) {
        return value;
    }

    const paddingNeeded = length - currentLength;
    const paddingString = (padChar || ' ').repeat(paddingNeeded);

    return paddingString + value;
};
```
