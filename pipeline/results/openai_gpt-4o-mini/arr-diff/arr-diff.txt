```javascript
module.exports = function(listA, listB) {
    if (listA === null) {
        return [];
    }
    if (listB === null) {
        return listA;
    }
    
    const result = [];
    const lookupSet = new Set(listB);
    
    for (const item of listA) {
        if (!lookupSet.has(item)) {
            result.push(item);
        }
    }

    return result;
};
```
