module.exports = function f(x) {
    if (x === null || typeof x === 'boolean' || typeof x === 'string' || typeof x === 'number' && !Number.isInteger(x)) {
        throw new TypeError("is-odd expects a number.");
    }

    if (typeof x !== 'number' || !Number.isInteger(x)) {
        throw new TypeError("is-odd expects a number.");
    }

    return x % 2 === 0;
};
