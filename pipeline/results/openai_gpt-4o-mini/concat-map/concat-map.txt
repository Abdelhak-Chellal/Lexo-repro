```javascript
module.exports = function(inputArray, funcString) {
    const func = eval(funcString); // Assuming safe and valid input
    const outputList = [];

    for (let i = 0; i < inputArray.length; i++) {
        try {
            const result = func(inputArray[i], i);
            if (Array.isArray(result)) {
                outputList.push(...result);
            } else {
                outputList.push(result);
            }
        } catch (e) {
            throw e; // Rethrow the exception for unhandled errors
        }
    }

    return outputList;
};
```
