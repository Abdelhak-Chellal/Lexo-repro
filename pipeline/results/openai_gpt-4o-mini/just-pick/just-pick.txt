```javascript
module.exports = function(dictionary, keys) {
    if (keys === null) {
        throw new TypeError("Cannot read properties of null (reading 'length')");
    }

    const result = {};

    if (typeof keys === 'string') {
        keys = [keys];
    }

    for (const key of keys) {
        if (key in dictionary) {
            result[key] = dictionary[key];
        }
    }

    return result;
};
```
