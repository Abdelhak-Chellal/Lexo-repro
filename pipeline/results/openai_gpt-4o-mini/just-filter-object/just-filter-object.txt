module.exports = function(inputDict, functionStr) {
    const conditionFunction = new Function('key', 'value', `return ${functionStr}`); // Create a callable function from the string

    const result = {};
    for (const key in inputDict) {
        const value = inputDict[key];
        if (conditionFunction(key, value)) {
            result[key] = value;
        }
    }
    
    return result;
}
