```javascript
module.exports = function(inputObject, funcString) {
    const func = eval(`(function(key, value) ${funcString})`);
    const outputObject = {};
    
    for (const [key, value] of Object.entries(inputObject)) {
        if (func(key, value)) {
            outputObject[key] = value;
        }
    }
    
    return outputObject;
};
```
