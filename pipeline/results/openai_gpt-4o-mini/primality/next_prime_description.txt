def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True

def nextprime(n):
    if not isinstance(n, int):
        raise TypeError(f"expect parameter n to be int. Given: {type(n)}.")
    
    if n <= 0:
        return 2
    
    current = n
    while True:
        if is_prime(current):
            return current
        current += 1
