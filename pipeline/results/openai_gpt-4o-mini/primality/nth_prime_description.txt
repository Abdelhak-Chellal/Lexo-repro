```python
def is_prime(num):
    if num < 2:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True

def f(n):
    primes = []
    candidate = 2
    while len(primes) <= n:
        if is_prime(candidate):
            primes.append(candidate)
        candidate += 1
    return primes[n]
```
