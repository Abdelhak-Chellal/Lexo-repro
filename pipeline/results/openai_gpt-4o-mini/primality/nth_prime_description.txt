def nth_prime(n):
    # Input validation
    if not isinstance(n, int):
        raise TypeError(f"nth_prime() takes (1) int type parameter. Given: {type(n).__name__}.")
    if n < 0:
        raise ValueError("nth_prime() int parameter must be positive or zero.")
    
    # Function to generate prime numbers up to a certain limit
    def generate_primes(limit):
        sieve = [True] * (limit + 1)  # true means prime
        sieve[0] = sieve[1] = False  # 0 and 1 are not primes
        for start in range(2, int(limit**0.5) + 1):
            if sieve[start]:
                for multiple in range(start*start, limit + 1, start):
                    sieve[multiple] = False
        return [num for num, is_prime in enumerate(sieve) if is_prime]

    # Estimate the limit for prime generation
    # A rough estimate can be taken as n * log(n) for larger n
    if n == 0:
        return 2
    limit = max(15, int(n * (log(n) if n > 1 else 1)))
    
    primes = generate_primes(limit)
    
    # Ensure we have enough primes
    while len(primes) <= n:
        limit *= 2
        primes = generate_primes(limit)
    
    return primes[n]
