def nth_prime(n):
    if n < 0:
        return None  # Handle edge case for negative input
    if n == 0:
        return 2  # Special case for f(0)
    
    limit = int(n * (log(n) + log(log(n)))) if n > 5 else 15  # Adjust for small n
    primes = sieve_of_eratosthenes(limit)
    
    # Ensure we have enough primes
    while len(primes) < n + 1:
        limit *= 2  # Increase limit and re-sieve
        primes = sieve_of_eratosthenes(limit)
    
    return primes[n]
