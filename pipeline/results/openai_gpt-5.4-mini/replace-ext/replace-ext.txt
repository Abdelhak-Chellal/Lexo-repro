module.exports = function replaceExt(path, ext) {
  if (typeof path !== 'string') return path;
  if (typeof ext !== 'string') ext = String(ext);

  const slashIndex = path.lastIndexOf('/');
  const dir = slashIndex === -1 ? '' : path.slice(0, slashIndex + 1);
  const base = slashIndex === -1 ? path : path.slice(slashIndex + 1);

  const dotIndex = base.lastIndexOf('.');

  if (dotIndex <= 0) {
    return path + ext;
  }

  return dir + base.slice(0, dotIndex) + ext;
};
