def is_prime(n):
    if type(n) is not int:
        raise TypeError(f"is_prime() takes (1) int type parameter. Given: {type(n)}.")
    if isinstance(n, bool):
        return False
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0:
        return False
    i = 3
    while i * i <= n:
        if n % i == 0:
            return False
        i += 2
    return True
