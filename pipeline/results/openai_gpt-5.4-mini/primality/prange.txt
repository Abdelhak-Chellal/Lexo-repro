def prange(n):
    if not isinstance(n, int):
        raise TypeError(f"prange() takes (1) int type parameter. Given: {type(n)}.")
    if n < 0:
        raise ValueError("prange() int parameter must be positive or zero.")
    if n == 0:
        return []

    primes = []
    candidate = 2

    while len(primes) < n:
        is_p = True
        for p in primes:
            if p * p > candidate:
                break
            if candidate % p == 0:
                is_p = False
                break
        if is_p:
            primes.append(candidate)
        candidate += 1

    return primes
