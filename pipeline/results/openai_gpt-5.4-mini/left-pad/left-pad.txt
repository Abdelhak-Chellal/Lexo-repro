module.exports = function (value, length, padding) {
  var str;
  if (value === null) {
    str = 'null';
  } else if (value === undefined) {
    str = 'undefined';
  } else if (value === false) {
    str = 'false';
  } else {
    str = String(value);
  }

  if (typeof length !== 'number' || !isFinite(length) || length < 0 || Math.floor(length) !== length) {
    return str;
  }

  if (str.length >= length) {
    return str;
  }

  var pad = padding === undefined ? ' ' : String(padding);
  if (pad.length === 0) {
    return str;
  }

  var needed = length - str.length;
  var repeated = '';
  while (repeated.length < needed) {
    repeated += pad;
  }

  return repeated.slice(0, needed) + str;
};
