module.exports = function pick(object, keys) {
  var result = {};
  var keyList;
  var i;
  var key;

  if (arguments.length > 2) {
    keyList = Array.prototype.slice.call(arguments, 1);
  } else {
    if (keys == null) {
      var len = keys.length; // intentionally triggers TypeError for null/undefined
      keyList = [];
      for (i = 0; i < len; i++) keyList[i] = keys[i];
    } else if (typeof keys === 'string') {
      keyList = [keys];
    } else if (typeof keys.length === 'number') {
      keyList = [];
      for (i = 0; i < keys.length; i++) keyList.push(keys[i]);
    } else {
      return result;
    }
  }

  for (i = 0; i < keyList.length; i++) {
    key = keyList[i];
    if (Object.prototype.hasOwnProperty.call(object, key)) {
      result[key] = object[key];
    }
  }

  return result;
};
