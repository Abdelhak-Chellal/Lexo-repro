module.exports = function (obj, keys) {
  var requestedKeys;

  if (arguments.length === 2 && Array.isArray(keys)) {
    requestedKeys = keys;
  } else {
    requestedKeys = Array.prototype.slice.call(arguments, 1);
  }

  var result = {};

  for (var i = 0; i < requestedKeys.length; i++) {
    var key = requestedKeys[i];

    if (typeof key !== 'string') {
      continue;
    }

    if (key in obj) {
      result[key] = obj[key];
    }
  }

  return result;
};
