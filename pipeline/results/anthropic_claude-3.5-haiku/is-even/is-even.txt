module.exports = function(x) {
    if (x === null || typeof x !== 'number' && typeof x !== 'string') {
        throw new Error('is-odd expects a number.');
    }
    
    try {
        const num = Number(x);
        
        if (isNaN(num)) {
            throw new Error('is-odd expects a number.');
        }
        
        return num % 2 === 0;
    } catch {
        throw new Error('is-odd expects a number.');
    }
};
