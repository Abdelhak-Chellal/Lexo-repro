module.exports = function(n) {
    if (n === null || n === undefined || typeof n === 'boolean' || Array.isArray(n) || typeof n === 'object') {
        throw new Error('is-odd expects a number');
    }
    
    if (typeof n === 'string') {
        try {
            n = parseInt(n, 10);
        } catch {
            throw new Error('is-odd expects a number');
        }
    }
    
    if (!Number.isInteger(n)) {
        throw new Error('is-odd expects an integer');
    }
    
    return n % 2 === 0;
};
