```javascript
module.exports = function(value) {
    if (value === null || typeof value === 'boolean') {
        return false;
    }
    
    if (typeof value !== 'string') {
        value = String(value);
    }
    
    value = value.trim();
    
    if (!value) {
        return false;
    }
    
    try {
        const num = parseFloat(value);
        return !isNaN(num) && isFinite(num);
    } catch {
        return false;
    }
};
```
