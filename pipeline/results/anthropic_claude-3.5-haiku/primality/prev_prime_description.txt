```python
def f(n):
    # If n is prime, return n-1 prime
    # If n is not prime, return largest prime <= n
    
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(x**0.5) + 1):
            if x % i == 0:
                return False
        return True
    
    # Start from n and go down until a prime is found
    for i in range(n, 1, -1):
        if is_prime(i):
            return i
    
    return 2  # Default case
```
