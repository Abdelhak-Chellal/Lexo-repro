def nth_prime(n):
    primes = [2]
    candidate = 3
    while len(primes) <= n:
        is_prime_flag = True
        for p in primes:
            if p * p > candidate:
                break
            if candidate % p == 0:
                is_prime_flag = False
                break
        if is_prime_flag:
            primes.append(candidate)
        candidate += 2
    return primes[n]
