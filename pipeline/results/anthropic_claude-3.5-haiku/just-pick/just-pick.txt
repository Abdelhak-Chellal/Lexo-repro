module.exports = function(sourceDict, keys) {
    if (typeof keys === 'string') {
        keys = [keys];
    }
    
    if (!keys.length) {
        return {};
    }
    
    return [...new Set(keys)].reduce((result, key) => {
        if (key in sourceDict) {
            result[key] = sourceDict[key];
        }
        return result;
    }, {});
};
