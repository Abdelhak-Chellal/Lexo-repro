module.exports = function(sourceDict, ...keys) {
    if (keys.length === 1 && Array.isArray(keys[0])) {
        keys = keys[0];
    }
    
    if (keys.length === 0) {
        return {};
    }
    
    const result = {};
    for (const key of keys) {
        if (Object.prototype.hasOwnProperty.call(sourceDict, key) && !result.hasOwnProperty(key)) {
            result[key] = sourceDict[key];
        }
    }
    
    return result;
};
