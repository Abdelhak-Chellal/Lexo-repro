module.exports = function(filename, new_ext) {
    if (filename === null) return null;
    if (typeof filename !== 'string') return filename;
    if (filename === '') return '';

    // Split into directory and basename
    let lastSep = Math.max(filename.lastIndexOf('/'), filename.lastIndexOf('\\'));
    let dirPart = '';
    let basename = filename;
    if (lastSep !== -1) {
        dirPart = filename.substring(0, lastSep + 1);
        basename = filename.substring(lastSep + 1);
    }

    // Strip trailing dots
    let stripped = basename.replace(/\.+$/, '');

    // Find last dot in stripped basename
    let lastDot = stripped.lastIndexOf('.');

    let namePart;
    if (lastDot > 0) {
        namePart = stripped.substring(0, lastDot);
    } else {
        namePart = stripped;
    }

    if (new_ext === '') {
        return dirPart + namePart;
    } else {
        return dirPart + namePart + new_ext;
    }
};
