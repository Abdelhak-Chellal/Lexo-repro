function f(inputArray, callbackStr) {
  const callback = new Function('x', 'i', callbackStr.replace(/^function\s*\([^)]*\)\s*\{/, '').replace(/\}$/, ''));
  
  const result = [];
  
  for (let i = 0; i < inputArray.length; i++) {
    const element = inputArray[i];
    const transformed = callback(element, i);
    
    if (Array.isArray(transformed)) {
      for (let j = 0; j < transformed.length; j++) {
        result.push(transformed[j]);
      }
    } else {
      result.push(transformed);
    }
  }
  
  return result;
}

module.exports = f;
