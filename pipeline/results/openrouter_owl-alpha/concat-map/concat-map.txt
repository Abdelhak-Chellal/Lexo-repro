module.exports = function(arr, funcStr) {
  var callback = eval('(' + funcStr + ')');
  var result = [];
  for (var i = 0; i < arr.length; i++) {
    var val = callback(arr[i], i);
    if (Array.isArray(val)) {
      result = result.concat(val);
    } else {
      result.push(val);
    }
  }
  return result;
}
