def prange(n):
    if not isinstance(n, int):
        raise TypeError("prange() takes (1) int type parameter. Given: <class 'float'>.")
    if n < 0:
        raise ValueError("prange() int parameter must be positive or zero.")
    if n == 0:
        return []
    primes = []
    candidate = 2
    while len(primes) < n:
        is_prime = True
        for prime in primes:
            if candidate % prime == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(candidate)
        candidate += 1
    return primes
