def nth_prime(n):
    if type(n) is not int:
        raise TypeError(f"nth_prime() takes (1) int type parameter. Given: {type(n)}.")
    if n < 0:
        raise ValueError("nth_prime() int parameter must be positive or zero.")
    
    if n < 6:
        upper_bound = 15
    else:
        import math
        upper_bound = int(n * (math.log(n) + math.log(math.log(n)))) + 10
    
    while True:
        is_prime = [True] * (upper_bound + 1)
        is_prime[0] = False
        is_prime[1] = False
        for i in range(2, int(upper_bound**0.5) + 1):
            if is_prime[i]:
                for j in range(i*i, upper_bound + 1, i):
                    is_prime[j] = False
        primes = [i for i in range(upper_bound + 1) if is_prime[i]]
        if len(primes) > n:
            return primes[n]
        upper_bound *= 2
