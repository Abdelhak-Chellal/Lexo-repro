function f(...args) {
  if (args.length === 1) {
    const x = args[0];
    if (Array.isArray(x)) {
      return x;
    } else {
      return [x];
    }
  } else {
    const first = args[0];
    if (Array.isArray(first) && first.length === 0) {
      return [];
    } else {
      return [first];
    }
  }
}

module.exports = f;
