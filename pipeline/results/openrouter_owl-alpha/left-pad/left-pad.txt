module.exports = function(value, width, padChar) {
  // Coerce value to string
  var s;
  if (value === null || value === undefined) {
    s = 'null';
  } else if (typeof value === 'boolean') {
    s = value.toString().toLowerCase();
  } else {
    s = String(value);
  }

  // Resolve padChar
  if (padChar === null || padChar === undefined || padChar === '') {
    padChar = ' ';
  }

  // Coerce padChar to string
  padChar = String(padChar);

  // If width is not larger than string length, return as-is
  if (width <= s.length) {
    return s;
  }

  // Build padding
  var paddingNeeded = width - s.length;
  var fullPad = padChar.repeat(Math.ceil(paddingNeeded / padChar.length));
  var padding = fullPad.substring(0, paddingNeeded);

  return padding + s;
};
