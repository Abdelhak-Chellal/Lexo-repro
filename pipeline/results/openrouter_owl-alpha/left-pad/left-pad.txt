module.exports = function(value, width, padChar) {
    let strValue;
    if (value === null || value === undefined) {
        strValue = 'null';
    } else if (typeof value === 'boolean') {
        strValue = value.toString();
    } else {
        strValue = String(value);
    }

    let effectivePad;
    if (padChar === null || padChar === undefined || padChar === '' || padChar === false) {
        effectivePad = ' ';
    } else {
        effectivePad = String(padChar);
    }

    if (width <= strValue.length) {
        return strValue;
    }

    const paddingNeeded = width - strValue.length;

    let padding;
    if (effectivePad.length === 1) {
        padding = effectivePad.repeat(paddingNeeded);
    } else {
        let repeated = '';
        while (repeated.length < paddingNeeded) {
            repeated += effectivePad;
        }
        padding = repeated.substring(0, paddingNeeded);
    }

    return padding + strValue;
};
