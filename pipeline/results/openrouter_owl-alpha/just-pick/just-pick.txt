module.exports = function(source, ...args) {
  if (source === null || source === undefined) {
    throw new TypeError("Cannot use 'in' operator to search for '" + (args[0] && args[0][0] || args[0]) + "' in null");
  }
  
  let keys;
  if (args.length === 1 && Array.isArray(args[0])) {
    keys = args[0];
  } else {
    keys = args;
  }
  
  const result = {};
  for (const key of keys) {
    if (key in source) {
      result[key] = source[key];
    }
  }
  
  return result;
};
