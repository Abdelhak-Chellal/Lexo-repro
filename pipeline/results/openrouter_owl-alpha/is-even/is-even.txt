module.exports = function(x) {
  if (typeof x === 'boolean') {
    throw new Error('is-odd expects a number.');
  }
  
  if (typeof x === 'string') {
    const num = Number(x);
    if (isNaN(num) || !Number.isInteger(num)) {
      throw new Error('is-odd expects a number.');
    }
    return num % 2 === 0;
  }
  
  if (typeof x === 'number') {
    if (!Number.isInteger(x)) {
      throw new Error('is-odd expects an integer.');
    }
    return x % 2 === 0;
  }
  
  throw new Error('is-odd expects a number.');
};
