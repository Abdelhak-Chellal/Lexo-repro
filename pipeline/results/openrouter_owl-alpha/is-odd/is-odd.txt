function f(n) {
  if (n === null || n === undefined) return false;
  if (Array.isArray(n)) return false;
  if (typeof n === 'object') throw new Error('expected a number');
  if (typeof n === 'string') throw new Error('expected a number');
  if (typeof n === 'boolean') return n === true;
  if (typeof n !== 'number') throw new Error('expected a number');
  if (!Number.isInteger(n)) throw new Error('expected an integer');
  if (!Number.isSafeInteger(n)) throw new Error('value exceeds maximum safe integer');
  return n % 2 !== 0 && n !== 0;
}

module.exports = f;
