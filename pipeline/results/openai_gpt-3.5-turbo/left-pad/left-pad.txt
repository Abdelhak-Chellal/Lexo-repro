```javascript
module.exports = function(input_string, length, character) {
    if (input_string === '' || length === 0) {
        return '';
    }
    if (length <= input_string.length) {
        return input_string;
    }
    const padding = length - input_string.length;
    const padded_string = (character || ' ').repeat(padding) + input_string;
    return padded_string;
};
```
