module.exports = function(obj, keys) {
    const result = {};
    if (!keys || keys.length === 0) {
        return result;
    }
    if (!Array.isArray(keys)) {
        keys = [keys];
    }
    for (const key in obj) {
        if (keys.includes(key)) {
            result[key] = obj[key];
        }
    }
    return result;
};
