module.exports = function f(d, keys) {
    let result_dict = {};
    
    if (keys === null || keys.length === 0) {
        return result_dict;
    }
    
    keys.forEach(key => {
        if (d.hasOwnProperty(key)) {
            result_dict[key] = d[key];
        }
    });
    
    return result_dict;
};
