module.exports = function(file_path, extension) {
    if (file_path === null) {
        return null;
    }
    if (extension === null) {
        extension = 'null';
    }
    
    let split_path = file_path.split('/');
    let file_name = split_path[split_path.length - 1];
    
    return split_path.slice(0, split_path.length - 1).join('/') + '/' + file_name + extension;
};
