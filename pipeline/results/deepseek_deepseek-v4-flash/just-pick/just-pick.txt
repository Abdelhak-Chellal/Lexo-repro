module.exports = function(obj, ...args) {
    if (obj === null || obj === undefined) {
        throw new TypeError("Cannot read properties of null (reading 'length')");
    }
    
    if (args.length === 0) {
        throw new TypeError("Cannot read properties of undefined (reading 'length')");
    }
    
    let keys;
    if (args.length === 1 && Array.isArray(args[0])) {
        keys = args[0];
    } else {
        keys = args;
    }
    
    const result = {};
    for (let key of keys) {
        if (key === null) {
            if ('null' in obj) {
                result['null'] = obj['null'];
            }
        } else if (typeof key === 'number' || typeof key === 'boolean') {
            const strKey = String(key);
            if (strKey in obj) {
                result[strKey] = obj[strKey];
            }
        } else if (key in obj) {
            result[key] = obj[key];
        }
    }
    
    return result;
};
