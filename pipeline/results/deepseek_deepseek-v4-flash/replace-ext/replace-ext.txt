module.exports = function(path, newExtension) {
  if (typeof path !== 'string') {
    return path;
  }

  if (path === '') {
    return '';
  }

  let processedPath = path;
  if (processedPath.endsWith('/') && processedPath !== '/') {
    processedPath = processedPath.slice(0, -1);
  }

  const lastSlashIndex = processedPath.lastIndexOf('/');
  const lastDotIndex = processedPath.lastIndexOf('.');

  if (lastDotIndex > lastSlashIndex) {
    if (newExtension === '') {
      return processedPath.slice(0, lastDotIndex);
    } else {
      return processedPath.slice(0, lastDotIndex) + newExtension;
    }
  } else {
    return processedPath + newExtension;
  }
};
