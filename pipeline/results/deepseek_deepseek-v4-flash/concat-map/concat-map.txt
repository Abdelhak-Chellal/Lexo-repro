```javascript
module.exports = function(array, callback) {
  const result = [];
  
  for (let i = 0; i < array.length; i++) {
    const value = callback(array[i], i);
    
    if (Array.isArray(value)) {
      for (let j = 0; j < value.length; j++) {
        result.push(value[j]);
      }
    } else {
      result.push(value);
    }
  }
  
  return result;
};
```
