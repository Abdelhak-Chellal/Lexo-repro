```javascript
module.exports = function(s, n, pad) {
  if (n <= s.length) {
    return s;
  }
  let fill;
  if (pad === undefined || pad === null || pad === false || pad === '') {
    fill = ' ';
  } else {
    fill = String(pad);
  }
  const count = n - s.length;
  const padding = fill.repeat(count);
  return padding + s;
};
```
